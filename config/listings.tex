% === ESTILO PADRÃO ===========================================================
\lstdefinestyle{mystyle}{
  basicstyle=\normalfont\ttfamily,
  numbers=left,
  numberstyle=\scriptsize,
  stepnumber=1,
  numbersep=1pt,
  showstringspaces=false,
  breaklines=true,
  postbreak=\mbox{\textcolor{red}{$\hookrightarrow$}\space},
  frame=lines,
  backgroundcolor=\color{background}
}

% === JSON ====================================================================

    basicstyle=\ttfamily\small\color{vscText},
    backgroundcolor=\color{vscBackground},
    stringstyle=\color{vscString},
    commentstyle=\color{vscComment},
    numberstyle=\color{vscNumber},
    morestring=[b]",
    morecomment=[l]{//},

% Realce especial: propriedades (antes dos :)
\makeatletter
\lst@AddToHook{OutputOther}{%
  \ifnum\lst@mode=\lst@Pmode
    \ifx\lst@token\lst@output
      \edef\lst@temp{\the\lst@token}%
      \ifnum\pdfstrcmp{\lst@temp}{\"}=0
        \def\lst@thestyle{\color{vscProperty}}%
      \fi
    \fi
  \fi
}
\makeatother
